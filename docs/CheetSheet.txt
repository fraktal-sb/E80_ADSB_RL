--------------------------------------------------------------------------------
stratuxrun kopieren/aktivieren
--------------------------------------------------------------------------------
sudo systemctl stop stratux
sudo overlayctl unlock

sudo cp /home/pi/stratuxrun /overlay/robase/opt/stratux/bin/stratuxrun
cp 80-e80-adsb.rules /etc/udev/rules.d/80-e80-adsb.rules

sudo chown root:root /overlay/robase/opt/stratux/bin/stratuxrun
sudo chmod 755 /overlay/robase/opt/stratux/bin/stratuxrun

sudo sync
sudo overlayctl lock
sudo systemctl start stratux

--------------------------------------------------------------------------------
Bluetooth deaktivieren (gpio + bluetooth zusammen geht nicht)
--------------------------------------------------------------------------------

sudo systemctl stop stratux
sudo overlayctl unlock
sudo nano /boot/firmware/config.txt
    dtoverlay=disable-bt
sudo systemctl disable hciuart
sudo overlayctl lock
sudo reboot
